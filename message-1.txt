#define num_of_module 6

typedef struct {
    char Module[20];
    float Mark;
    int coefficient;
}Grade;

typedef struct Student {
    long ID;
    char Last_name[50];
    char First_name[50];
    Grade mark_s[num_of_module];
    float average;
    struct Student *next;
    bool passed;
}Student;

Student* searchid(Student *head, long id) {
    while (head != NULL) {
        if (head->ID == id)
            {return head;}
        head = head->next;
    }
    return NULL;
}

float average_student(Student *s) {
    float somme = 0;
    int total_coef = 0;
    for (int i = 0; i < num_of_module; i++) {
        if (s->mark_s[i].Mark >= 0) {
        somme += s->mark_s[i].Mark * s->mark_s[i].coefficient;
        total_coef += s->mark_s[i].coefficient;
    }
}
    if (total_coef == 0) {
        s->average = 0;
    } else {
        s->average = somme / total_coef;
    }

    s->passed = (s->average >= 10);
    return s->average;
}

Student *deleteStudent(Student *head, long id) {
    Student *q = head;
    Student *p = NULL;

    while (q != NULL && q->ID != id) {
        p = q;
        q = q->next;
    }
    if (q == NULL) {
        printf("Student not found.\n");
        return head;
    }
    if (p == NULL) {
        head = q->next;
    } else {
        p->next = q->next;
    }
    free(q);
    printf("Student deleted.\n");
    return head;
}

Student *insertionSort(Student *head){
    Student *sortedHead = NULL;
    Student *student = head;

    while (student != NULL){
        Student *nextStudent = student->next;
        if (sortedHead == NULL || student->average >= sortedHead->average){
            student->next = sortedHead;
            sortedHead = student;
        }
        else{
            Student *position = sortedHead;
            while (position->next != NULL && position->next->average > student->average){
                position = position->next;
            }
            student->next = position->next;
            position->next = student;
        }
        student = nextStudent;
    }
    return sortedHead;
}

void Statistics(Student *head) {
    float total_average = 0.0f;
    int student_count = 0;
    int total_passes = 0;
    for (Student *student = head; student != NULL; student = student->next) {
        total_average += student->average;
        student_count++;
        if (student->average >= 10.0f) {
            total_passes++;
        }
    }

    printf("\nTotal number of students: %d\n", student_count);
    if (student_count == 0) {
        printf("No students.\n");
        printf("Pass rate: 0.00%%\n");
        return;
    }
    printf("Class average: %.2f\n", total_average / student_count);
    printf("Pass rate: %.2f%%\n",
           100.0f * total_passes / student_count);
}


void freeStudents(Student *head){
    Student *student_list = head;

    while (student_list != NULL){

        Student *nextStudent = student_list->next;

        free(student_list);

        student_list = nextStudent;

    }
}
